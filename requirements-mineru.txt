# The isolated environment for the MinerU preprocessing (scripts/run_mineru.py).
#
# **It cannot share a venv with the main package.** MinerU 3.x wants
# transformers>=4.57.3 and torch>=2.6,<3, while this package pins
# torch>=2.11,<2.12; the two do not resolve together. The Python versions leave no
# room either: the main package requires >=3.13 and MinerU requires <3.14.
#
# So the MinerU stage lives in its own venv (.venv-mineru) and hands its output to
# the pipeline (MinerUChunker) as Markdown and content_list.json.
#
# Set up:  bash scripts/setup_mineru_env.sh
# Run:     .venv-mineru/bin/python scripts/run_mineru.py ...
#
# The pipeline backend is the one used: dedicated models for layout, OCR,
# equations and tables. The vlm backend needs vllm, which is not worth the setup
# cost here.
mineru[pipeline]==3.4.3

# torch is pinned explicitly, inside mineru's own range (>=2.6,<3). Left
# unspecified, torch 2.13.0+cu130 arrives, does not match the nvidia-cudnn-cu13
# 9.20 that comes with it, and conv2d dies with
# CUDNN_STATUS_SUBLIBRARY_VERSION_MISMATCH. 2.8.0's PyPI default is the cu128
# build, which works on an RTX 3090 (sm_86).
torch==2.8.0
torchvision==0.23.0

# Imported by the pytorchocr bundled with mineru 3.4.3
# (mineru/model/utils/pytorchocr). It is missing from mineru's own dependency
# declaration, so it is added here.
six
